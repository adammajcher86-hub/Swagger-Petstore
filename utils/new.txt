When('User opens {string} suborder', { timeout: 60000 }, async function (locator: string) {
  console.log('=== [openSuborder] START ===');
  console.log('[openSuborder] raw locator string:', locator);

  let title: string;
  if (locator === 'Natco Installation Poland') title = 'natco_installation';
  else title = locator.toLowerCase().replaceAll(' ', '_');

  console.log('[openSuborder] computed title attribute:', title);

  // --- STEP 0: Log all open pages in context ---
  console.log('[openSuborder] STEP 0: listing all open pages in browserContext...');
  const allPages = this.browserContext.pages();
  console.log('[openSuborder] STEP 0: total open pages:', allPages.length);
  for (let i = 0; i < allPages.length; i++) {
    console.log(`[openSuborder] STEP 0: page[${i}] url:`, allPages[i].url());
    console.log(`[openSuborder] STEP 0: page[${i}] title:`, await allPages[i].title());
  }

  // --- STEP 0b: Log current this.page ---
  console.log('[openSuborder] STEP 0b: this.page url:', this.page.url());
  console.log('[openSuborder] STEP 0b: this.page title:', await this.page.title());

  // --- STEP 1: Find the correct page that has #taskstable ---
  console.log('[openSuborder] STEP 1: searching for page containing #taskstable across all open pages...');
  let targetPage = null;
  for (let i = 0; i < allPages.length; i++) {
    const p = allPages[i];
    const hasTasksTable = await p.evaluate(() => !!document.querySelector('#taskstable'));
    const hasTaskSuborders = await p.evaluate(() => !!document.querySelector('#tasksuborders'));
    const trTitles = await p.evaluate(() =>
      Array.from(document.querySelectorAll('tr[title]')).map(r => r.getAttribute('title'))
    );
    console.log(`[openSuborder] STEP 1: page[${i}] url=${p.url()}`);
    console.log(`[openSuborder] STEP 1: page[${i}] #taskstable exists: ${hasTasksTable}`);
    console.log(`[openSuborder] STEP 1: page[${i}] #tasksuborders exists: ${hasTaskSuborders}`);
    console.log(`[openSuborder] STEP 1: page[${i}] tr[title] values:`, JSON.stringify(trTitles));
    if (trTitles.includes(title)) {
      console.log(`[openSuborder] STEP 1: FOUND target row on page[${i}] ✓`);
      targetPage = p;
    }
  }

  if (!targetPage) {
    console.warn('[openSuborder] STEP 1: target row not found on any page yet - falling back to this.page');
    targetPage = this.page;
  }

  console.log('[openSuborder] STEP 1: using targetPage url:', targetPage.url());
  const page = targetPage;

  // --- STEP 2: Wait for spinner to disappear on target page ---
  console.log('[openSuborder] STEP 2: waiting for spinner #taskSuborderTabSpinner to be hidden...');
  try {
    await page.locator('#taskSuborderTabSpinner').waitFor({ state: 'hidden', timeout: 30000 });
    console.log('[openSuborder] STEP 2: spinner hidden ✓');
  } catch (e) {
    console.warn('[openSuborder] STEP 2: spinner wait failed or not found - continuing:', e.message);
  }

  // --- STEP 3: Dump all tr[title] on target page ---
  console.log('[openSuborder] STEP 3: dumping all tr[title] on targetPage...');
  const allTitles = await page.evaluate(() =>
    Array.from(document.querySelectorAll('tr[title]')).map(r => r.getAttribute('title'))
  );
  console.log('[openSuborder] STEP 3: tr[title] count:', allTitles.length);
  console.log('[openSuborder] STEP 3: tr[title] values:', JSON.stringify(allTitles));

  // --- STEP 4: Wait for target row to be attached ---
  console.log(`[openSuborder] STEP 4: waiting for tr[title="${title}"] to be attached...`);
  const rowLocator = page.locator(`tr[title="${title}"]`);
  const rowCountBefore = await rowLocator.count();
  console.log('[openSuborder] STEP 4: row count before waitFor:', rowCountBefore);

  try {
    await rowLocator.waitFor({ state: 'attached', timeout: 30000 });
    console.log('[openSuborder] STEP 4: row attached ✓');
  } catch (e) {
    console.error('[openSuborder] STEP 4: waitFor attached FAILED:', e.message);
    const tableHtml = await page.evaluate(() => {
      const t = document.querySelector('#taskstable') ?? document.querySelector('table');
      return t ? t.outerHTML.substring(0, 5000) : 'NO TABLE FOUND';
    });
    console.error('[openSuborder] STEP 4: table HTML dump:\n', tableHtml);
    throw e;
  }

  // --- STEP 5: Check row visibility ---
  console.log('[openSuborder] STEP 5: checking row visibility state...');
  const isVisible = await rowLocator.isVisible();
  const isHidden = await rowLocator.isHidden();
  console.log(`[openSuborder] STEP 5: isVisible=${isVisible}, isHidden=${isHidden}`);

  // --- STEP 6: scrollIntoView + getBoundingClientRect ---
  console.log('[openSuborder] STEP 6: scrollIntoView and getBoundingClientRect...');
  const coords = await page.evaluate((t) => {
    const row = document.querySelector(`tr[title="${t}"]`);
    if (!row) return null;
    row.scrollIntoView({ behavior: 'instant', block: 'center' });
    const rect = row.getBoundingClientRect();
    const style = window.getComputedStyle(row);
    return {
      x: rect.left + rect.width / 2,
      y: rect.top + rect.height / 2,
      rect: { top: rect.top, left: rect.left, width: rect.width, height: rect.height },
      display: style.display,
      visibility: style.visibility,
      opacity: style.opacity,
    };
  }, title);

  console.log('[openSuborder] STEP 6: coords + style:', JSON.stringify(coords));

  if (!coords) {
    const allTitlesAfter = await page.evaluate(() =>
      Array.from(document.querySelectorAll('tr[title]')).map(r => r.getAttribute('title'))
    );
    console.error('[openSuborder] STEP 6: coords null - tr[title] now:', JSON.stringify(allTitlesAfter));
    throw new Error(`Row not found for title: ${title}`);
  }

  // --- STEP 7: Viewport sanity check ---
  const viewportSize = page.viewportSize();
  console.log('[openSuborder] STEP 7: viewport:', JSON.stringify(viewportSize));
  console.log(`[openSuborder] STEP 7: dblclick target x=${coords.x}, y=${coords.y}`);
  if (coords.y < 0 || (viewportSize && coords.y > viewportSize.height)) {
    console.warn('[openSuborder] STEP 7: WARNING - y is outside viewport, scroll may not have worked');
  }
  if (coords.rect.width === 0 || coords.rect.height === 0) {
    console.warn('[openSuborder] STEP 7: WARNING - row has zero dimensions, likely not rendered');
  }

  // --- STEP 8: Settle + dblclick ---
  console.log('[openSuborder] STEP 8: waiting 400ms for scroll to settle...');
  await delay(400);

  const newTabPromise = this.browserContext.waitForEvent('page', { timeout: 30000 });
  console.log('[openSuborder] STEP 8: firing dblclick...');
  await page.mouse.dblclick(coords.x, coords.y);
  console.log('[openSuborder] STEP 8: dblclick fired, waiting for new tab...');

  // --- STEP 9: New tab ---
  let newTab;
  try {
    newTab = await newTabPromise;
    console.log('[openSuborder] STEP 9: new tab opened ✓ url:', await newTab.url());
  } catch (e) {
    console.error('[openSuborder] STEP 9: no new tab opened within 30s:', e.message);
    await page.screenshot({ path: 'debug-no-new-tab.png' });
    console.log('[openSuborder] STEP 9: screenshot saved → debug-no-new-tab.png');
    throw e;
  }

  // --- STEP 10: Load new tab ---
  console.log('[openSuborder] STEP 10: waiting for domcontentloaded...');
  await newTab.waitForLoadState('domcontentloaded', { timeout: 30000 });
  console.log('[openSuborder] STEP 10: new tab loaded, url:', await newTab.url());

  await newTab.bringToFront();
  this.newTab = newTab;
  this.page = newTab;

  console.log('=== [openSuborder] END - success ===');
});
