When('User opens {string} suborder', { timeout: 60000 }, async function (locator: string) {
  console.log('=== [openSuborder] START ===');
  console.log('[openSuborder] raw locator string:', locator);

  const page = this.page;
  console.log('[openSuborder] this.page url:', page.url());

  // --- STEP 1: Wait for spinner ---
  try {
    await page.locator('#taskSuborderTabSpinner').waitFor({ state: 'hidden', timeout: 15000 });
    console.log('[openSuborder] STEP 1: spinner hidden ✓');
  } catch (e) {
    console.warn('[openSuborder] STEP 1: spinner skip:', e.message);
  }

  // --- STEP 2: Locate row ---
  const rowLocator = page.locator('tr', { hasText: locator }).first();
  await rowLocator.waitFor({ state: 'attached', timeout: 30000 });
  console.log('[openSuborder] STEP 2: row attached ✓');

  // --- STEP 3: Deep inspect - find all anchors, buttons, clickable children ---
  console.log('[openSuborder] STEP 3: deep inspect row for links/buttons/click handlers...');
  const rowInspect = await rowLocator.evaluate((row) => {
    const anchors = Array.from(row.querySelectorAll('a')).map(a => ({
      tag: 'A', href: a.href, text: a.textContent?.trim(), id: a.id
    }));
    const buttons = Array.from(row.querySelectorAll('button')).map(b => ({
      tag: 'BUTTON', text: b.textContent?.trim(), id: b.id
    }));
    const allChildren = Array.from(row.querySelectorAll('*')).map(el => {
      const style = window.getComputedStyle(el);
      return {
        tag: el.tagName,
        id: el.id,
        className: el.className,
        cursor: style.cursor,
        hasClickListener: typeof (el as any).onclick === 'function',
        text: el.textContent?.trim().substring(0, 50),
      };
    }).filter(el => el.cursor === 'pointer' || el.tag === 'A' || el.tag === 'BUTTON');

    // Check if row itself has onclick
    const rowStyle = window.getComputedStyle(row);
    return {
      anchors,
      buttons,
      clickableChildren: allChildren,
      rowCursor: rowStyle.cursor,
      rowOnclick: typeof (row as any).onclick === 'function',
      rowHasEventListeners: row.getAttribute('onclick'),
    };
  });
  console.log('[openSuborder] STEP 3: row inspect:', JSON.stringify(rowInspect, null, 2));

  // --- STEP 4: Check if suborder ID can be used to construct direct URL ---
  const suborderId = await rowLocator.evaluate((row) => {
    const hiddenTd = row.querySelector('td[style*="display:none"]');
    return hiddenTd?.textContent?.trim() ?? null;
  });
  console.log('[openSuborder] STEP 4: suborder hidden ID:', suborderId);

  // Try direct URL navigation if we have an ID
  if (suborderId) {
    const currentUrl = page.url();
    const possibleUrls = [
      `${currentUrl}/suborders/${suborderId}`,
      `${currentUrl.replace(/\/$/, '')}/suborder/${suborderId}`,
      page.url().replace(/ngsd\/[^/]+/, `ngsd/${suborderId}`),
    ];
    console.log('[openSuborder] STEP 4: possible suborder URLs:', possibleUrls);
  }

  await delay(400);

  // --- STEP 5: Try dispatch native events directly on the row element ---
  console.log('[openSuborder] STEP 5: trying native dispatchEvent on row...');

  const tryGetNewTab = async (action: () => Promise<void>, label: string): Promise<any> => {
    const pagesBefore = this.browserContext.pages().length;
    const newTabPromise = this.browserContext.waitForEvent('page', { timeout: 6000 }).catch(() => null);
    await action();
    await delay(2000);
    const newTabResult = await newTabPromise;
    const pagesAfter = this.browserContext.pages();
    console.log(`[openSuborder] ${label}: before=${pagesBefore} after=${pagesAfter.length} newTabEvent=${!!newTabResult}`);
    if (newTabResult) return newTabResult;
    if (pagesAfter.length > pagesBefore) return pagesAfter[pagesAfter.length - 1];
    return null;
  };

  // Strategy A: dispatch dblclick event natively on the row
  let newTab = await tryGetNewTab(async () => {
    await rowLocator.evaluate((row) => {
      row.dispatchEvent(new MouseEvent('dblclick', { bubbles: true, cancelable: true }));
    });
  }, 'STEP 5A: native dblclick dispatchEvent');

  // Strategy B: dispatch click natively
  if (!newTab) {
    newTab = await tryGetNewTab(async () => {
      await rowLocator.evaluate((row) => {
        row.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true }));
      });
    }, 'STEP 5B: native click dispatchEvent');
  }

  // Strategy C: click the name TD specifically (not the whole row)
  if (!newTab) {
    const nameTd = page.locator('td', { hasText: locator }).first();
    newTab = await tryGetNewTab(
      () => nameTd.dblclick({ timeout: 5000 }),
      'STEP 5C: dblclick on name TD directly'
    );
  }

  // Strategy D: click name TD single click
  if (!newTab) {
    const nameTd = page.locator('td', { hasText: locator }).first();
    newTab = await tryGetNewTab(
      () => nameTd.click({ timeout: 5000 }),
      'STEP 5D: single click on name TD'
    );
  }

  // Strategy E: navigate directly to suborder URL
  if (!newTab && suborderId) {
    console.log('[openSuborder] STEP 5E: trying direct URL navigation with suborder ID...');
    const baseUrl = page.url().replace(/\/$/, '');
    const directUrl = `${baseUrl}/suborders/${suborderId}`;
    console.log('[openSuborder] STEP 5E: navigating to:', directUrl);
    newTab = await tryGetNewTab(async () => {
      await page.goto(directUrl);
    }, 'STEP 5E: direct URL navigation');
    if (!newTab) {
      // Maybe it navigated in same tab
      console.log('[openSuborder] STEP 5E: current url after goto:', page.url());
      if (page.url() !== baseUrl + '/') {
        console.log('[openSuborder] STEP 5E: same-tab navigation detected ✓');
        newTab = page;
      }
    }
  }

  if (!newTab) {
    await page.screenshot({ path: 'debug-no-new-tab.png' });
    console.error('[openSuborder] all strategies failed - screenshot saved');
    throw new Error(`Could not open suborder "${locator}" - all strategies exhausted`);
  }

  // --- STEP 6: Load ---
  console.log('[openSuborder] STEP 6: new tab url:', await newTab.url());
  await newTab.waitForLoadState('domcontentloaded', { timeout: 30000 });
  console.log('[openSuborder] STEP 6: loaded ✓');

  await newTab.bringToFront();
  this.newTab = newTab;
  this.page = newTab;

  console.log('=== [openSuborder] END - success ===');
});
