// Check if tr is inside a shadow root
const tr = document.querySelector('tr[title="natco_installation_slovakia"]');
console.log('found:', tr);
console.log('parent:', tr?.parentElement?.tagName);
console.log('grandparent:', tr?.parentElement?.parentElement?.tagName);
console.log('great-grandparent:', tr?.parentElement?.parentElement?.parentElement?.tagName);

// Walk up to find shadow host
let el = tr;
while (el) {
  if (el.getRootNode() instanceof ShadowRoot) {
    console.log('INSIDE SHADOW DOM of:', el.getRootNode().host.tagName, el.getRootNode().host.id);
    break;
  }
  el = el.parentElement;
}
