export const datePicker = async (page: Page, days: number, dateType: string) => {
  // --- Calculate target date ---
  const today = new Date();
  const targetDate = new Date(today.setDate(today.getDate() + days));
  const targetDay = targetDate.getDate();
  const targetDayString = targetDay < 10 ? ('0' + targetDay).toString() : targetDay.toString();
  const targetMonth = targetDate.toLocaleString('en-US', { month: 'long' });
  const targetMonthNo = String(targetDate.getMonth() + 1).padStart(2, '0');
  const targetYear = targetDate.getFullYear().toString().trim();
  const currentYear = new Date().getFullYear().toString().trim();

  console.log(`[datePicker] days=${days}, dateType=${dateType}`);
  console.log(`[datePicker] targetDay=${targetDayString}, targetMonth=${targetMonth}, targetMonthNo=${targetMonthNo}, targetYear=${targetYear}`);

  const { container } = getDateSelectors(dateType);
  console.log(`[datePicker] container selector: ${container}`);

  // --- Wait for dialog: try chained locator (shadow DOM pierce) first,
  //     fall back to legacy CSS string (works when no shadow root) ---
  console.log('[datePicker] waiting for duet-date-picker dialog...');

  const chainedDialog = page
    .locator(container)
    .locator('duet-date-picker')
    .locator('.duet-date__dialog[aria-hidden="false"]');

  const legacyDialog = page.locator(
    `${container} duet-date-picker .duet-date__dialog[aria-hidden="false"]`
  );

  let useShadowPierce: boolean;
  try {
    await chainedDialog.waitFor({ state: 'visible', timeout: 10000 });
    useShadowPierce = true;
    console.log('[datePicker] dialog visible via chained locator (shadow DOM) ✓');
  } catch {
    console.warn('[datePicker] chained locator timed out - trying legacy CSS selector...');
    await legacyDialog.waitFor({ state: 'visible', timeout: 10000 });
    useShadowPierce = false;
    console.log('[datePicker] dialog visible via legacy CSS selector ✓');
  }

  // --- Helper to build correct locator based on which strategy worked ---
  const pickerLocator = (innerSelector: string) => {
    if (useShadowPierce) {
      return page
        .locator(container)
        .locator('duet-date-picker')
        .locator(innerSelector);
    }
    return page.locator(`${container} duet-date-picker ${innerSelector}`);
  };

  // --- Select month ---
  console.log(`[datePicker] selecting month: ${targetMonth}`);
  await pickerLocator('.duet-date__select--month').selectOption({ label: targetMonth });
  console.log('[datePicker] month selected ✓');

  // --- Select year (only if different from current year) ---
  if (targetYear !== currentYear) {
    console.log(`[datePicker] selecting year: ${targetYear}`);
    await pickerLocator('.duet-date__select--year').selectOption({ label: targetYear });
    console.log('[datePicker] year selected ✓');
  } else {
    console.log(`[datePicker] year unchanged (${targetYear}), skipping`);
  }

  // --- Click target day button ---
  const dayButtonName = `${targetDayString} ${targetMonth}`;
  console.log(`[datePicker] waiting for day button: "${dayButtonName}"`);
  const dayButton = page.getByRole('button', { name: dayButtonName, exact: true });
  await expect(dayButton).toBeEnabled({ timeout: 5000 });
  console.log('[datePicker] day button enabled ✓, clicking...');
  await dayButton.click();
  console.log('[datePicker] day clicked ✓');

  const formattedDate = `${targetYear}-${targetMonthNo}-${targetDayString}`;
  console.log(`[datePicker] returning formattedDate: ${formattedDate}`);
  return formattedDate;
};
