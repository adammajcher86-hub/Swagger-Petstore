    for (let i = 1; i <= rows.length; i++) {
      const innerLocators = await page.locator(LOCATORS.struncturedInfoRows(i)).all();
      const actualValues = await Promise.all(
        innerLocators.map(async (innerLocator) => {
          const val = await innerLocator.getAttribute('value');
          return val?.trim() || '';
        }),
      );
      const [actualType, actualValue] = actualValues;
      const expectedType = records[i - 1]['Field'];
      let expectedValue = records[i - 1]['Value'];
      if (expectedValue === 'ngsdID') expectedValue = this.orderId;
      if (expectedType === 'Customer/Contract-No') {
        expect(actualValue.slice(0, -3)).toEqual(expectedValue.slice(0, -3));
      } else {
        expect(actualType).toEqual(expectedType);
        expect(actualValue).toContain(expectedValue);
      }
    }
    return;
