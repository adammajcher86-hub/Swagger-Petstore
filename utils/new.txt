When('User opens {string} suborder', { timeout: 60000 }, async function (locator: string) {
  const page = this.page;
  await delay(5000);
  let title: string;
  if (locator === 'Natco Installation Poland') title = 'natco_installation';
  else title = locator.toLowerCase().replaceAll(' ', '_');

  console.log('....User opens {string} suborder: begin of method');
  console.log('title:', title);

  const newTabPromise = this.browserContext.waitForEvent('page', { timeout: 30000 });

  const row = page.locator(`tr[title='${title}']`);

  console.log('scrolling into view...');
  await row.scrollIntoViewIfNeeded({ timeout: 15000 });
  await delay(500);

  console.log('clicking first time...');
  await row.click({ force: true });
  await delay(500);
  console.log('clicking second time...');
  await row.click({ force: true });

  console.log('waiting for new tab...');
  const newTab = await newTabPromise;
  console.log('new tab url:', await newTab.url());

  await newTab.waitForLoadState('domcontentloaded', { timeout: 30000 });
  await newTab.bringToFront();
  this.newTab = newTab;
  this.page = newTab;
});
