// Check event listeners on the tr and its parents
const row = document.querySelector('tr[title="natco_installation_slovakia"]');
const td = row.querySelector('td[title]') || row.querySelectorAll('td')[1];

// Get all ancestors up to body
let el = td;
while (el && el !== document.body) {
  const listeners = getEventListeners(el);
  if (Object.keys(listeners).length > 0) {
    console.log('listeners on:', el.tagName, el.id, el.className, listeners);
  }
  el = el.parentElement;
}
