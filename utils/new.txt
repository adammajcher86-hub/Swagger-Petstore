When('User opens {string} suborder', { timeout: 60000 }, async function (locator: string) {
  console.log('=== [openSuborder] START ===');
  console.log('[openSuborder] raw locator string:', locator);

  const page = this.page;
  console.log('[openSuborder] this.page url:', page.url());

  // --- STEP 1: Wait for spinner ---
  console.log('[openSuborder] STEP 1: waiting for spinner to hide...');
  try {
    await page.locator('#taskSuborderTabSpinner').waitFor({ state: 'hidden', timeout: 15000 });
    console.log('[openSuborder] STEP 1: spinner hidden ✓');
  } catch (e) {
    console.warn('[openSuborder] STEP 1: spinner not found or already hidden:', e.message);
  }

  // --- STEP 2: Find row by text content ---
  // The table uses <td style="text-align:left;">Natco Installation Slovakia</td>
  // with NO title on <tr>, so we locate by visible text inside the row
  console.log(`[openSuborder] STEP 2: locating row containing text "${locator}"...`);

  // Find the <tr> that contains a <td> with exact text matching locator
  const rowLocator = page.locator('tr', { hasText: locator }).first();

  // --- STEP 3: Wait for row attached ---
  console.log('[openSuborder] STEP 3: waiting for row to be attached...');
  try {
    await rowLocator.waitFor({ state: 'attached', timeout: 30000 });
    const count = await rowLocator.count();
    console.log('[openSuborder] STEP 3: row attached ✓, count:', count);
  } catch (e) {
    console.error('[openSuborder] STEP 3: row not found by text:', e.message);
    // Dump all td texts for debugging
    const allTdTexts = await page.evaluate(() =>
      Array.from(document.querySelectorAll('td[style*="text-align:left"], td[style*="text-align: left"]'))
        .map(td => td.textContent?.trim())
        .filter(Boolean)
    );
    console.error('[openSuborder] STEP 3: all left-aligned td texts:', JSON.stringify(allTdTexts));
    throw e;
  }

  // --- STEP 4: Check visibility ---
  const isVisible = await rowLocator.isVisible();
  console.log(`[openSuborder] STEP 4: row isVisible=${isVisible}`);

  // --- STEP 5: scrollIntoView + coords via JS ---
  console.log('[openSuborder] STEP 5: scrollIntoView + getBoundingClientRect...');
  const coords = await page.evaluate((text) => {
    // Find all tds with matching text, get their parent tr
    const tds = Array.from(document.querySelectorAll('td'));
    const matchingTd = tds.find(td => td.textContent?.trim() === text);
    if (!matchingTd) {
      console.log('[evaluate] no td found with text:', text);
      return null;
    }
    const row = matchingTd.closest('tr');
    if (!row) {
      console.log('[evaluate] td found but no parent tr');
      return null;
    }
    row.scrollIntoView({ behavior: 'instant', block: 'center' });
    const rect = row.getBoundingClientRect();
    const style = window.getComputedStyle(row);
    return {
      x: rect.left + rect.width / 2,
      y: rect.top + rect.height / 2,
      rect: { top: rect.top, left: rect.left, width: rect.width, height: rect.height },
      display: style.display,
      visibility: style.visibility,
      opacity: style.opacity,
    };
  }, locator);

  console.log('[openSuborder] STEP 5: coords + style:', JSON.stringify(coords));

  if (!coords) {
    const allTexts = await page.evaluate(() =>
      Array.from(document.querySelectorAll('td'))
        .map(td => td.textContent?.trim())
        .filter(Boolean)
    );
    console.error('[openSuborder] STEP 5: coords null - all td texts:', JSON.stringify(allTexts));
    throw new Error(`Row not found for text: ${locator}`);
  }

  // --- STEP 6: Viewport sanity check ---
  const viewportSize = page.viewportSize();
  console.log('[openSuborder] STEP 6: viewport:', JSON.stringify(viewportSize));
  console.log(`[openSuborder] STEP 6: dblclick target x=${coords.x}, y=${coords.y}`);
  if (coords.rect.width === 0 || coords.rect.height === 0) {
    console.warn('[openSuborder] STEP 6: WARNING - zero dimensions on row');
  }
  if (coords.y < 0 || (viewportSize && coords.y > viewportSize.height)) {
    console.warn('[openSuborder] STEP 6: WARNING - y outside viewport');
  }

  // --- STEP 7: Settle + dblclick ---
  console.log('[openSuborder] STEP 7: waiting 400ms then dblclick...');
  await delay(400);

  const newTabPromise = this.browserContext.waitForEvent('page', { timeout: 30000 });
  console.log('[openSuborder] STEP 7: firing dblclick at', coords.x, coords.y);
  await page.mouse.dblclick(coords.x, coords.y);
  console.log('[openSuborder] STEP 7: dblclick fired, waiting for new tab...');

  // --- STEP 8: New tab ---
  let newTab;
  try {
    newTab = await newTabPromise;
    console.log('[openSuborder] STEP 8: new tab opened ✓ url:', await newTab.url());
  } catch (e) {
    console.error('[openSuborder] STEP 8: no new tab within 30s:', e.message);
    await page.screenshot({ path: 'debug-no-new-tab.png' });
    console.log('[openSuborder] STEP 8: screenshot saved → debug-no-new-tab.png');
    throw e;
  }

  // --- STEP 9: Load new tab ---
  console.log('[openSuborder] STEP 9: waiting for domcontentloaded...');
  await newTab.waitForLoadState('domcontentloaded', { timeout: 30000 });
  console.log('[openSuborder] STEP 9: loaded, url:', await newTab.url());

  await newTab.bringToFront();
  this.newTab = newTab;
  this.page = newTab;

  console.log('=== [openSuborder] END - success ===');
});
