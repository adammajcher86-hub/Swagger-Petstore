// Check what happens when we manually dblclick via JS
const row = document.querySelector('tr[title="natco_installation_slovakia"]');
row.addEventListener('dblclick', (e) => console.log('dblclick fired!', e));
row.dispatchEvent(new MouseEvent('dblclick', { bubbles: true, cancelable: true }));

// Also check what event actually opens the tab
const table = document.querySelector('#tasksuborders');
['click', 'dblclick', 'mousedown', 'mouseup'].forEach(evt => {
  table.addEventListener(evt, (e) => console.log(evt, 'on:', e.target.tagName, e.target.title));
});
