When('User opens {string} suborder', { timeout: 60000 }, async function (locator: string) {
  const page = this.page;

  try {
    await page.locator('#taskSuborderTabSpinner').waitFor({ state: 'hidden', timeout: 15000 });
  } catch {
    // already hidden or not present
  }

  const rowLocator = page.locator('tr', { hasText: locator }).first();
  await rowLocator.waitFor({ state: 'attached', timeout: 30000 });

  await rowLocator.evaluate((el) => {
    (el as HTMLElement).scrollIntoView({ behavior: 'instant', block: 'center' });
  });
  await delay(300);

  const newTabPromise = this.browserContext.waitForEvent('page', { timeout: 15000 });

  // Primary: native dispatchEvent (works in headed mode)
  await rowLocator.evaluate((el) => {
    (el as HTMLElement).dispatchEvent(
      new MouseEvent('dblclick', { bubbles: true, cancelable: true })
    );
  });

  // If no new tab within 5s, fall back to Playwright dblclick (isTrusted=true, works headless)
  const newTab = await newTabPromise.catch(async () => {
    console.warn('[openSuborder] dispatchEvent had no effect - falling back to Playwright dblclick');
    const fallbackPromise = this.browserContext.waitForEvent('page', { timeout: 15000 });
    await rowLocator.dblclick({ timeout: 5000 });
    return fallbackPromise;
  });

  await newTab.waitForLoadState('domcontentloaded', { timeout: 30000 });
  await newTab.bringToFront();
  this.newTab = newTab;
  this.page = newTab;
});
