Then('User {string} on {string} button', async function (action: string, locatorValue: string) {
  // Reduced from 500ms — just enough to let UI settle after previous step
  await delay(100);

  if (action === 'double clicks') {
    if (locatorValue === 'order ID') locatorValue = this.orderId;
    this.mainPage = this.page;

    const start = Date.now();
    const [newTab] = await Promise.all([
      this.page.context().waitForEvent('page'),
      this.page.getByText(locatorValue).last().dblclick()
    ]);

    // 'domcontentloaded' is faster than 'load' — sufficient for most cases
    await newTab.waitForLoadState('domcontentloaded');
    const end = Date.now();
    console.log(`Order details page loading time: ${end - start} ms`);
    await newTab.bringToFront();
    this.newTab = newTab;
    this.page = newTab;

  } else if (locatorValue.includes('endpoint-B')) {
    const page = this.page;
    const endpoint = locatorValue.split(' ');
    // Removed redundant toBeVisible() — click() auto-waits for actionability
    await page.getByRole('button', { name: endpoint[1], exact: true }).last().click();

  } else if (locatorValue.includes('close icon')) {
    const page = this.page;
    const locator = LOCATORS[locatorValue] || 'button[part="close-button"][aria-label="Close"]';

    let attempts = 0;
    let closed = false;
    while (attempts < 3) {
      const closeBtns = page.locator(locator);
      const count = await closeBtns.count();
      if (count === 0) break;
      for (let i = 0; i < count; i++) {
        const btn = closeBtns.nth(i);
        if (await btn.isVisible()) {
          try {
            await btn.click({ timeout: 5000 });
            // Reduced from 500ms — just enough for modal close animation
            await page.waitForTimeout(200);
            closed = true;
          } catch (e) {
            // Ignore and try next
          }
        }
      }
      const stillVisible = await closeBtns.filter({ has: page.locator('text=Close') }).count();
      if (stillVisible === 0) break;
      attempts++;
    }
    if (!closed) {
      throw new Error('Could not close the popup after multiple attempts');
    }

  } else {
    const page = this.page;
    // Removed redundant toBeVisible() check — click() has built-in auto-waiting
    await page.getByRole('button', { name: locatorValue, exact: true }).first().click();
    // Replaced hardcoded delay(3000) with smart network wait
    await waitForPageLoad(page);
  }

  await waitForPageLoad(this.page);
});





async function waitForPageLoad(page: Page, timeout = 10000): Promise<void> {
  await Promise.race([
    page.waitForLoadState('networkidle', { timeout }),
    page.waitForLoadState('load', { timeout }),
  ]).catch(() => {}); // Best effort — don't fail the step on timeout
}
