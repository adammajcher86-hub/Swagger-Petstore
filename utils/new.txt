export async function verifyScaleTextFieldValue(
  page: Page,
  expectedValue: string,
  id?: string | null,
  label?: string | null,
  name?: string | null,
): Promise<boolean> {
  let locator;

  if (id) {
    locator = page.locator(`#${id}`);
  } else {
    locator = label ? page.getByLabel(label) : page.locator('scale-text-field');
    if (name) {
      locator = locator.and(page.locator(`[name="${name}"]`));
    }
  }

  console.log(`[verifyScaleTextField] id="${id}" label="${label}" name="${name}" expected="${expectedValue}"`);

  try {
    // Try 1: el.value (works for native inputs)
    let actualValue = await locator.evaluate((el) => (el as HTMLInputElement).value).catch(() => null);
    console.log(`[verifyScaleTextField] el.value="${actualValue}"`);

    // Try 2: shadow root input value (scale-text-field wraps input in shadow DOM)
    if (actualValue === null || actualValue === undefined || actualValue === '') {
      actualValue = await locator.evaluate((el) => {
        const shadow = (el as HTMLElement).shadowRoot;
        const input = shadow?.querySelector('input') ?? shadow?.querySelector('textarea');
        return (input as HTMLInputElement)?.value ?? null;
      }).catch(() => null);
      console.log(`[verifyScaleTextField] shadow input value="${actualValue}"`);
    }

    // Try 3: getAttribute('value')
    if (actualValue === null || actualValue === undefined || actualValue === '') {
      actualValue = await locator.getAttribute('value').catch(() => null);
      console.log(`[verifyScaleTextField] getAttribute value="${actualValue}"`);
    }

    // Try 4: innerText
    if (actualValue === null || actualValue === undefined || actualValue === '') {
      actualValue = await locator.innerText().catch(() => null);
      console.log(`[verifyScaleTextField] innerText="${actualValue}"`);
    }

    const result = actualValue === expectedValue;
    console.log(`[verifyScaleTextField] result=${result} actual="${actualValue}" expected="${expectedValue}"`);
    return result;

  } catch (error) {
    console.error(`[verifyScaleTextField] error:`, (error as Error).message);
    return false;
  }
}
