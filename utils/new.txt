// 1. Sprawdź czy element w ogóle istnieje i jaki ma dokładnie atrybut name
const el = document.querySelector('scale-text-field[name="sno"]');
console.log('Element znaleziony:', el);
console.log('Atrybut name:', el?.getAttribute('name'));

// 2. Sprawdź czy ma open shadow root
console.log('Shadow root:', el?.shadowRoot);

// 3. Sprawdź input wewnątrz shadow roota
const input = el?.shadowRoot?.querySelector('input');
console.log('Input:', input);
console.log('Input value:', input?.value);
console.log('Input name (wewnętrzny):', input?.getAttribute('name'));
