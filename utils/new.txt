When('Verify status of the tasks & Suborders:', async function (dataTable: DataTable) {
  const page = this.page;
  await waitForPageLoad(page);
  await selectTab(page, 'Tasks & SubOrders');

  const tableData = dataTable.hashes();

  for (const data of tableData) {
    const taskName = data['Task Name'];
    const expectedStatus = data['Status'];

    console.log(`[verifyStatus] waiting for "${taskName}" to have status "${expectedStatus}"...`);

    // Poll instead of fixed delay - wait up to 30s for status to match
    await expect(
      page.locator(LOCATORS.statusOfTaskOrSuborder(taskName))
    ).toContainText(expectedStatus, { timeout: 30000 });

    console.log(`[verifyStatus] "${taskName}" = "${expectedStatus}" ✓`);
  }
});
