document.querySelectorAll('duet-date-picker').forEach((picker, i) => {
  const container = picker.closest('[id]');
  console.log(`Picker ${i}: container id = ${container?.id}, parent id = ${picker.parentElement?.id}`);
});

document.querySelectorAll('duet-date-picker').forEach(picker => {
  let el = picker;
  while (el) {
    if (el.id) console.log('Found id:', el.id, 'tagName:', el.tagName);
    el = el.parentElement;
  }
});
